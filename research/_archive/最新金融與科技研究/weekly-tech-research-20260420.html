4.5 GiS 立即可用的 Python 範本

# 用法：對 LLM 研報評分做 conformal prediction，widths ≥4 送人工 import numpy as np from math import ceil # 1. 用 200 筆人工標註建校準集 nonconf = [abs(llm_score - human_score) for llm_score, human_score in calib] n = len(nonconf); alpha = 0.10 q_hat = np.quantile(nonconf, ceil((1 - alpha) * (n + 1)) / n) # 2. 推論：回傳預測集而非單分數 def predict_set(llm_score): return [y for y in range(1, 6) if abs(llm_score - y) <= q_hat] # 3. 部署決策 c = predict_set(new_llm_score) if len(c) <= 2: action = "採納" # 窄預測集 → 可信 elif len(c) >= 4: action = "送人工複審" # 寬預測集 → 警告 else: action = "標記中等信心"
