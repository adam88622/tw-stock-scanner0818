import pandas as pd
df = pd.read_parquet(r'D:\claude\tw-stock-scanner\data\institutional_full.parquet')
# 篩近一年外資買超累積前 20
recent = df[df['date'] >= df['date'].max() - pd.Timedelta('365D')]
top = recent.groupby(['stock_id','name'])['foreign_net'].sum().nlargest(20)
print(top)
